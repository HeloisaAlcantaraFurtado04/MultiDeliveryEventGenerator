
BASE_TIME_GRANULARITY = 1MINUTE
START = 0,
END = 720MINUTE


# Database tables
RESOURCES
TABLES
   users, # student and driver
   sellers, # stores involved
   items, # which products stores sell
   orders, # connect student, driver, delivery
   order_items, # connect individual products to one order
   events, # represent activity occurring
END TABLES

VARIABLES
END VARIABLES

FUNCTIONS
END FUNCTIONS


# A student starts an order after signing in
CREATE EVENTTYPE OrderStarted (. #beginning of the shopping process 
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: AUTO_INCREMENT,
   student_id: orders(student_id) FOREIGNKEY, #existing order from the student id
   delivery_location: orders(delivery_location),
   estimated_delivery_time: orders(estimated_delivery_time),
   created_at: orders(created_at),

   order_id -> ALL
)


# The student adds items from different sellers
CREATE EVENTTYPE AddItem (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_item_id: AUTO_INCREMENT, # generates a distintic id for each order item 
   order_id: INHERIT OrderStarted(order_id), #reuses the one that it started 
   item_id: items(item_id) FOREIGNKEY,
   seller_id: items(seller_id) FOREIGNKEY, 
   quantity: order_items(quantity),

   order_item_id -> ALL # it relates to all attributes 
)


# The student places the order after adding items
CREATE EVENTTYPE OrderPlaced (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderStarted(order_id), #reuses the original order id
   student_id: INHERIT OrderStarted(student_id), #same student 
   delivery_location: INHERIT OrderStarted(delivery_location),
   estimated_delivery_time: INHERIT OrderStarted(estimated_delivery_time),#reuses the delivery estimate
   created_at: INHERIT OrderStarted(created_at),

   order_id -> ALL
)


# Payment is successful
CREATE EVENTTYPE PaymentCompleted (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id), # id of the confirmed order 
   student_id: INHERIT OrderPlaced(student_id), #student id of the same order-placement process 

   order_id -> ALL
)


# Payment fails
CREATE EVENTTYPE PaymentFailed (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id),
   student_id: INHERIT OrderPlaced(student_id),

   order_id -> ALL
)


# The order is cancelled after a failed payment
CREATE EVENTTYPE OrderCancelled (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentFailed(order_id),# payment fails leads to order cancelled 
   student_id: INHERIT PaymentFailed(student_id),

   order_id -> ALL
)


# A seller starts preparing their items
CREATE EVENTTYPE ItemsPreparing (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentCompleted(order_id), #only if payment succeeds 
   seller_id: sellers(seller_id) FOREIGNKEY,
   is_available: items(is_available), #checks if the product is available 

   (order_id, seller_id) -> ALL
)


# A seller finishes preparing their items
CREATE EVENTTYPE ItemsReady (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT ItemsPreparing(order_id),
   seller_id: INHERIT ItemsPreparing(seller_id),

   (order_id, seller_id) -> ALL
)


# A driver is assigned to the order
CREATE EVENTTYPE DriverAssigned (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentCompleted(order_id),
   driver_id: orders(driver_id) FOREIGNKEY,

   order_id -> driver_id
)


# Driver picks up items from a seller
CREATE EVENTTYPE ItemsPickedUp (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT ItemsReady(order_id),
   seller_id: INHERIT ItemsReady(seller_id),
   driver_id: INHERIT DriverAssigned(driver_id),

   (order_id, seller_id) -> driver_id
)


# The complete order reaches the student
CREATE EVENTTYPE OrderDelivered (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),

   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id),
   student_id: INHERIT OrderPlaced(student_id),
   driver_id: INHERIT DriverAssigned(driver_id),
   delivery_location: INHERIT OrderPlaced(delivery_location),
   estimated_delivery_time: INHERIT OrderPlaced(estimated_delivery_time),

   order_id -> ALL
)


# ----------------------------------
# EVENT RELATIONSHIPS
# ----------------------------------

# A student starts an order
E0: CAUSE START; OrderStarted


# The student adds between 1 and 5 items
# Adding items takes 1 to 5 minutes
E1: CAUSE OrderStarted; AddItem
   TIME BETWEEN 1MINUTE AND 5MINUTE
   COUNT BETWEEN 1 AND 5


# The student reviews the cart and places the order
# Order placement happens 6 to 10 minutes after starting
E2: CAUSE OrderStarted; OrderPlaced
   TIME BETWEEN 6MINUTE AND 10MINUTE


# Payment happens after the order is placed
# We assume this takes 4 to 9 minutes from order placement
E3: CAUSE OrderPlaced; PaymentCompleted
   TIME BETWEEN 4MINUTE AND 9MINUTE


# Each seller starts preparing their part of the order
# We simulate between 1 and 4 sellers
E4: CAUSE PaymentCompleted; ItemsPreparing
   TIME BETWEEN 1MINUTE AND 4MINUTE
   COUNT BETWEEN 1 AND 4


# Preparation takes different amounts of time
# The distribution for E5 is defined below
E5: CAUSE ItemsPreparing; ItemsReady


# A driver is assigned while the sellers are preparing
# We assume driver assignment takes 3 to 6 minutes
E6: CAUSE PaymentCompleted; DriverAssigned
   TIME BETWEEN 3MINUTE AND 6MINUTE


# The driver travels to the seller and picks up the items
# We assume this takes 6 to 13 minutes after items are ready
E7: CAUSE ItemsReady; ItemsPickedUp
   TIME BETWEEN 6MINUTE AND 13MINUTE


# Approximate delivery timing after payment
# A longer delivery time is possible
# Distribution for E8 is defined below
#
# NOTE: This is a timing approximation.
# It does not check whether every seller has been picked up.
E8: CAUSE PaymentCompleted; OrderDelivered


# If a payment failure is generated, cancel the order
# Payment failure branching is not implemented yet
E9: CAUSE PaymentFailed; OrderCancelled
   TIME BETWEEN 1MINUTE AND 3MINUTE


TERMINATE OrderDelivered
TERMINATE OrderCancelled


# ----------------------------------
# ORDER ARRIVAL DISTRIBUTION
# ----------------------------------

# We simulate a 720-minute campus day.
# For this example, minute 0 represents 8 AM.
#
# Fewer orders happen in the morning and afternoon.
# We expect more orders around lunch and dinner.
#
# The values are the number of simulated orders
# in each interval, not percentages.

CREATE DISTRIBUTION FOR START (
   BASE_TIME_GRANULARITY = 1MINUTE,
   [0,120) : 8,
   [120,240) : 18,
   [240,360) : 36,
   [360,480) : 12,
   [480,600) : 22,
   [600,720) : 40
)


# ----------------------------------
# PREPARATION TIME DISTRIBUTION
# ----------------------------------

# We assume most sellers finish preparing
# their items in 4 to 8 minutes.
#
# Some sellers take 9 to 15 minutes,
# depending on the items and workload.
#
# These are initial simulation assumptions.

CREATE TIME DISTRIBUTION FOR E5 (
   BASE_TIME_GRANULARITY = 1MINUTE,
   [4,9) : 70,
   [9,16) : 30
)


# ----------------------------------
# DELIVERY TIME DISTRIBUTION
# ----------------------------------

# These times are measured from PaymentCompleted.
#
# We assume most orders finish within
# 35 to 49 minutes after payment.
#
# Some orders take 50 to 79 minutes
# because of longer pickup or travel times.
#
# The minimum is set after the maximum
# estimated pickup time in this simplified model.
#
# This is not yet a true ETA-based delay check.

CREATE TIME DISTRIBUTION FOR E8 (
   BASE_TIME_GRANULARITY = 1MINUTE,
   [35,50) : 85,
   [50,80) : 15
)

END
