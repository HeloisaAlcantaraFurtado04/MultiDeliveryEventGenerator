
BASE_TIME_GRANULARITY = 1 MINUTE
START = 0,
END = 720 MINUTE

RESOURCES
TABLES
users, sellers, items, orders, order_items, events, 
END TABLES
VARIABLES
END VARIABLES
FUNCTIONS
END FUNCTIONS


CREATE EVENTTYPE OrderStarted ( 
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: AUTO_INCREMENT,
   student_id: orders(student_id) FOREIGNKEY, 
   delivery_location: orders(delivery_location),
   estimated_delivery_time: orders(estimated_delivery_time),
   created_at: orders(created_at),
   order_id -> ALL
)

CREATE EventType AddItem (
	EVENTTIME  1,
	INGESTTIME (1) MAXDELAY (10),
	event_id: AUTO_INCREMENT,
	order_item_id: AUTO_INCREMENT,
	order_id: INHERIT OrderStarted(order_id),
	item_id: Item(item_id) FOREIGNKEY,
	seller_id: Item(seller_id) FOREIGNKEY,
	quantity: OrderItem(quantity),
	order_item_id -> ALL
)


CREATE EventType OrderPlaced (
	EVENTTIME  1,
	INGESTTIME (1) MAXDELAY (10),
	event_id: AUTO_INCREMENT,
	order_id: INHERIT OrderStarted(order_id),
	student_id: INHERIT OrderStarted(student_id),
	delivery_location: INHERIT OrderStarted(delivery_location),
	estimated_delivery_time: INHERIT OrderStarted(estimated_delivery_time),
	created_at: INHERIT OrderStarted(created_at),
	order_id -> ALL
)


CREATE EVENTTYPE PaymentCompleted (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id), 
   student_id: INHERIT OrderPlaced(student_id),
   order_id -> ALL
)


CREATE EVENTTYPE PaymentFailed (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id),
   student_id: INHERIT OrderPlaced(student_id),
   order_id -> ALL
)

CREATE EVENTTYPE OrderCancelled (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentFailed(order_id),
   student_id: INHERIT PaymentFailed(student_id),
   order_id -> ALL
)


CREATE EVENTTYPE ItemsPreparing (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentCompleted(order_id), 
   seller_id: sellers(seller_id) FOREIGNKEY,
   is_available: items(is_available), 
   (order_id, seller_id) -> ALL
)

CREATE EVENTTYPE ItemsReady (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT ItemsPreparing(order_id),
   seller_id: INHERIT ItemsPreparing(seller_id),
   (order_id, seller_id) -> ALL  
)

CREATE EVENTTYPE DriverAssigned (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT PaymentCompleted(order_id),
   driver_id: orders(driver_id) FOREIGNKEY, 
   order_id -> driver_id
)

CREATE EVENTTYPE ItemsPickedUp (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT ItemsReady(order_id),
   seller_id: INHERIT ItemsReady(seller_id),
   driver_id: INHERIT DriverAssigned(driver_id),
   (order_id, seller_id) -> driver_id
)

CREATE EVENTTYPE OrderDelivered (
   EVENTTIME 1,
   INGESTTIME (1) MAXDELAY (10),
   event_id: AUTO_INCREMENT,
   order_id: INHERIT OrderPlaced(order_id), 
   student_id: INHERIT OrderPlaced(student_id),
   driver_id: INHERIT DriverAssigned(driver_id),
   delivery_location: INHERIT OrderPlaced(delivery_location),
   estimated_delivery_time: INHERIT OrderPlaced(estimated_delivery_time),
   order_id -> ALL
)


E0: CAUSE START; OrderStarted
E1: CAUSE OrderStarted; AddItem TIME BETWEEN 1min AND 5min
E2: CAUSE OrderStarted; OrderPlaced TIME BETWEEN 6min AND 10min
E3: CAUSE OrderPlaced; PaymentCompleted TIME BETWEEN 4min AND 9min
E4: CAUSE PaymentCompleted; ItemsPreparing TIME BETWEEN 1min AND 4min
E5: CAUSE ItemsPreparing; ItemsReady
E6: CAUSE PaymentCompleted; DriverAssigned TIME BETWEEN 3min AND 6min
E7: CAUSE ItemsReady; ItemsPickedUp TIME BETWEEN 6min AND 13min
E8: CAUSE ItemsPickedUp; OrderDelivered
E9: CAUSE PaymentFailed; OrderCancelled TIME BETWEEN 1min AND 3min
E10: CAUSE OrderPlaced; PaymentFailed TIME BETWEEN 4min AND 9min
TERMINATE OrderDelivered
TERMINATE OrderCancelled
